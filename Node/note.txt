HTTP Module
**********************

To create a server you need node http module

import http from 'node:http'
()
make sure you add 
type: module in package.json

create server

const server = http.create(Server(req, res)=>{
    res.end("hello from server)
})

server.listen(8000, ()=> `server running on port ${PORT}`)

const server = http.createServer( (req, res)=> {
  res.write('This is some data \n')
  res.write('This is some more data \n')
  res.end('This is from the server', 'utf8', ()=> console.log('response end'))
})

server.listen(PORT, () => console.log(`Connected on port: ${PORT}`))

you must always res.end() this will always finish the response

#### res.writeHead can be used instead of res.setHeader
  res.setHeader('Content-Type', 'application/json')
  res.setHeader('Access-Control-Allow-Origin', '*') // allow access from any domain
  res.setHeader('Access-Control-Allow-Methods', 'GET') // GET method only

  The only caveat is writeHead stops any further modificaiton to the request, nothing else
  after it with res.setHeader or writeHead works. 

  Just stick to res.setHeader excep you have an explicit reason

fs.readFile(path) --------: Async but Too much code also not encouraged
const server = http.createServer((req, res) => {

  const pathToResource = path.join(__dirname, 'public', 'index.html')

  fs.readFile(pathToResource, 'utf8', (err, content) => {
    if (err) {
      console.log(err)
      return
    }
    res.statusCode = 200
    res.setHeader('Content-Type', 'text/html')
    res.end(content)
  })

})

******************************
Events
An event signals that something happened
Maybe a user clicked a button
file finished downloading
A network request is recieved. etc.
**********

// import EventEmitter
import { EventEmitter } from 'node:events'

const customerDetails = {
  fullName: 'Meryl Sheep',
  email: 'baah@thedevilwearswool.com',
  phone: 12345678910
}

// create the emitter
const emailRequestEmitter = new EventEmitter()

// define the listener function
function generateEmail(customer) {
  console.log(`Email generated for ${customer.email}`)
}

// register the listener
emailRequestEmitter.on('emailRequest', generateEmail)
emailRequestEmitter.on('emailRequest', () => console.log('task assigned'))
emailRequestEmitter.on('emailRequest', () => console.log('email logged'))

// emit the event
setTimeout(()=> {
  emailRequestEmitter.emit('emailRequest', customerDetails)
}, 2000)
